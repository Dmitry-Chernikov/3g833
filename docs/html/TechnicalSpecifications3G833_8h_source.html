64 // Коэффициенты редукции шкива от двигателя к входному валу
65 constexpr double PULLEY_RATIO_1 = static_cast<double>(GEAR_PULLEY_SPEED_ONE1) / GEAR_PULLEY_SPEED_ONE2; // Первая передача 70/190 = 0.36
66 constexpr double PULLEY_RATIO_2 = static_cast<double>(GEAR_PULLEY_SPEED_TWO1) / GEAR_PULLEY_SPEED_TWO2; // Вторая передача 84/175 = 0.48
67 constexpr double PULLEY_RATIO_3 = static_cast<double>(GEAR_PULLEY_SPEED_THREE1) / GEAR_PULLEY_SPEED_THREE2; // Третья передача 110/155 = 0.71
68
69 // Параметры зубчатых передач в редукторе
70 constexpr int GEAR_Z1 = 18; // Число зубьев шестерни на валу 1
71 constexpr int GEAR_Z2 = 45; // Число зубьев шестерни на валу 2 (первая пара)
72 constexpr int GEAR_Z3 = 22; // Число зубьев шестерни на валу 2 (вторая пара)
73 constexpr int GEAR_Z4 = 55; // Число зубьев шестерни на валу 3 (вторая пара)
74
75 // Выходная шестерня на валу 3
76 constexpr int GEAR_Z5 = 17; // Число зубьев выходной шестерни
77 constexpr double MODULE_MM = 3.0; // Модуль зубьев, мм
78
79 // Расчёт передаточных отношений редуктора
80 constexpr double GEAR_RATIO_1 = static_cast<double>(GEAR_Z1) / GEAR_Z2; // 18/45 = 0.4
81 constexpr double GEAR_RATIO_2 = static_cast<double>(GEAR_Z3) / GEAR_Z4; // 22/55 = 0.4
82 constexpr double GEARBOX_RATIO = GEAR_RATIO_1 * GEAR_RATIO_2; // 0.4*0.4 = 0.16
83
84 // Диаметр делительной окружности выходной шестерни, мм
85 constexpr double PITCH_DIAMETER_MM = MODULE_MM * GEAR_Z5; // 3*17 = 51 мм
86
87 // Перемещение рейки за один оборот шестерни, мм
88 constexpr double TRAVEL_PER_REVOLUTION_MM = PI * PITCH_DIAMETER_MM; // π*51 ≈ 160.221 мм
89
90 // Константа для расчёта скорости рейки
91 constexpr double SPEED_CONSTANT = (motorReturnsRPM * TRAVEL_PER_REVOLUTION_MM * GEARBOX_RATIO) / 1000.0; // (930*160.221*0.16) ≈ 23.840 м/мин
92
93 // Финальные скорости возвратно-поступательного движения, м/мин

95 constexpr double CALC_RETURNS_SPEED_2 = SPEED_CONSTANT * PULLEY_RATIO_2; // (23.840*0.48) ≈ 11.443 м/мин
96 constexpr double CALC_RETURNS_SPEED_3 = SPEED_CONSTANT * PULLEY_RATIO_3; // (23.840*0.71) ≈ 16.927 м/мин
